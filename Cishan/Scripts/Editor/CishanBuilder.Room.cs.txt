        const float RoomX = WorkshopLayout.RoomX;
        const float RoomZ = WorkshopLayout.RoomZ;
        const float Eave = WorkshopLayout.Eaves;
        const float RidgeY = WorkshopLayout.Ridge;
        const float Wall = WorkshopLayout.WallT;

        static void BuildRoom()
        {
            Transform room = Group("01_棚屋", root, Vector3.zero, Vector3.zero).transform;

            // interior stone floor
            Mesh floorMesh = ProcMesh.Plane(RoomX, RoomZ, 1f / 2.4f);
            GameObject floor = MeshObj("石地", room, floorMesh, mats.floor,
                new Vector3(RoomX * 0.5f, 0f, RoomZ * 0.5f), Vector3.zero, Vector3.one);
            BoxCollider floorCol = floor.AddComponent<BoxCollider>();
            floorCol.size = new Vector3(RoomX, 0.2f, RoomZ);
            floorCol.center = new Vector3(0f, -0.1f, 0f);
            floor.isStatic = true;

            BuildSouthFrame(room);
            BuildNorthWall(room);
            BuildWestWall(room);
            BuildEastWall(room);
            BuildRoof(room);
            BuildNorthWindows(room);
        }

        static void BuildNorthWall(Transform room)
        {
            Transform g = Group("北墙", room, Vector3.zero, Vector3.zero).transform;
            GameObject wall = BoxObj("灰墙_北", g, new Vector3(RoomX + Wall * 2f, Eave, Wall),
                new Vector3(RoomX * 0.5f, Eave * 0.5f, RoomZ + Wall * 0.5f),
                mats.plaster, Vector3.zero, 0.01f, true);
            wall.isStatic = true;

            Vector2[] gable =
            {
                new Vector2(-RoomX * 0.5f, 0f),
                new Vector2(RoomX * 0.5f, 0f),
                new Vector2(0f, RidgeY - Eave)
            };
            Mesh mesh = ProcMesh.Extrude(gable, 0.1f, 0.4f);
            MeshObj("山墙_北", g, mesh, mats.plaster,
                new Vector3(RoomX * 0.5f, Eave, RoomZ - 0.01f), Vector3.zero, Vector3.one);

            BoxObj("墙帽", g, new Vector3(13.1f, 0.12f, 0.55f), new Vector3(6.3f, 4.2f, 7.2f),
                mats.woodDark, Vector3.zero, 0.01f, false);
        }

        static void BuildRoof(Transform room)
        {
            Transform g = Group("屋面", room, Vector3.zero, Vector3.zero).transform;
            float slope = Mathf.Atan2(RidgeY - Eave, RoomZ * 0.5f) * Mathf.Rad2Deg;

            BoxObj("瓦面_南", g, new Vector3(13.2f, 0.12f, 3.95f),
                new Vector3(6.3f, 4.72f, 1.78f), mats.roof,
                new Vector3(-slope, 0f, 0f), 0.01f, true);
            BoxObj("瓦面_北", g, new Vector3(13.2f, 0.12f, 3.95f),
                new Vector3(6.3f, 4.72f, 5.42f), mats.roof,
                new Vector3(slope, 0f, 0f), 0.01f, true);
            BoxObj("脊梁", g, new Vector3(13.1f, 0.2f, 0.34f), new Vector3(6.3f, 5.17f, 3.6f),
                mats.woodDark, Vector3.zero, 0.01f, true);

            float[] xs = { 0.8f, 2.6f, 4.4f, 6.2f, 8.0f, 9.8f, 11.8f };
            for (int i = 0; i < xs.Length; i++)
            {
                BoxObj("椽_南_" + i, g, new Vector3(0.11f, 0.15f, 3.9f),
                    new Vector3(xs[i], 4.58f, 1.78f), mats.wood,
                    new Vector3(-slope, 0f, 0f), 0.008f, false);
                BoxObj("椽_北_" + i, g, new Vector3(0.11f, 0.15f, 3.9f),
                    new Vector3(xs[i], 4.58f, 5.42f), mats.wood,
                    new Vector3(slope, 0f, 0f), 0.008f, false);
            }

            for (int i = 0; i < 3; i++)
            {
                float z = 0.95f + i * 1.3f;
                float y = Mathf.Lerp(Eave, RidgeY, z / (RoomZ * 0.5f)) - 0.3f;
                BoxObj("檩_南_" + i, g, new Vector3(13.0f, 0.12f, 0.12f), new Vector3(6.3f, y, z),
                    mats.wood, Vector3.zero, 0.008f, false);
                BoxObj("檩_北_" + i, g, new Vector3(13.0f, 0.12f, 0.12f), new Vector3(6.3f, y, RoomZ - z),
                    mats.wood, Vector3.zero, 0.008f, false);
            }

            for (int i = 0; i < 3; i++)
            {
                float x = 2.6f + i * 3.7f;
                BoxObj("大梁_" + i, g, new Vector3(0.16f, 0.22f, 7.2f), new Vector3(x, 4.06f, 3.6f),
                    mats.woodDark, Vector3.zero, 0.012f, true);
            }
        }

        static void BuildLighting()
        {
            Transform g = Group("05_光", root, Vector3.zero, Vector3.zero).transform;

            GameObject sunGO = new GameObject("主光_南侧棚口");
            sunGO.transform.SetParent(g, false);
            sunGO.transform.position = new Vector3(2f, 10f, -9f);
            sunGO.transform.LookAt(new Vector3(8f, 0f, 3.5f));
            Light sun = sunGO.AddComponent<Light>();
            sun.type = LightType.Directional;
            sun.color = new Color(1f, 0.965f, 0.91f);
            sun.intensity = 1.15f;
            sun.shadows = LightShadows.Soft;
            sun.shadowStrength = 0.78f;
            sun.shadowBias = 0.04f;
            sun.shadowNormalBias = 0.35f;
            sun.shadowResolution = LightShadowResolution.VeryHigh;
            RenderSettings.sun = sun;

            Point("补光_室内", g, new Vector3(5.2f, 3.0f, 2.6f), new Color(0.85f, 0.82f, 0.76f), 1.6f, 12f);
            Point("窗光_北", g, new Vector3(6.3f, 3.1f, 6.5f), new Color(0.94f, 0.90f, 0.81f), 1.1f, 6.5f);
            Point("水光_轮坑", g, new Vector3(11.9f, 1.1f, -1.3f), new Color(0.62f, 0.70f, 0.69f), 0.8f, 4.5f);
            Point("入口光", g, new Vector3(0.4f, 2.2f, 3.9f), new Color(0.90f, 0.85f, 0.78f), 0.8f, 5f);
        }

        static Light Point(string name, Transform parent, Vector3 pos, Color color,
            float intensity, float range)
        {
            GameObject go = new GameObject(name);
            go.transform.SetParent(parent, false);
            go.transform.position = pos;
            Light l = go.AddComponent<Light>();
            l.type = LightType.Point;
            l.color = color;
            l.intensity = intensity;
            l.range = range;
            l.shadows = LightShadows.None;
            return l;
        }
