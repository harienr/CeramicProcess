        static void BuildDryingArea()
        {
            Transform g = Group("15_晾干入仓", root, Vector3.zero, Vector3.zero).transform;
            Vector3 c = new Vector3(1.15f, 0f, 2.35f);

            for (int s = -1; s <= 1; s += 2)
            {
                BoxObj("晾架柱_" + s, g, new Vector3(0.08f, 1.4f, 0.08f),
                    new Vector3(c.x + s * 0.85f, 0.7f, c.z), mats.wood,
                    new Vector3(0f, 0f, -s * 8f), 0.006f, false);
            }
            float[] ys = { 0.42f, 0.78f, 1.12f };
            for (int i = 0; i < ys.Length; i++)
            {
                BoxObj("晾板_" + i, g, new Vector3(1.8f, 0.03f, 0.48f),
                    new Vector3(c.x, ys[i], c.z), mats.wood, Vector3.zero, 0.004f, false);
                for (int k = 0; k < 3; k++)
                {
                    Brick(g, new Vector3(c.x - 0.55f + k * 0.55f, ys[i] + 0.04f, c.z),
                        new Vector3(0f, k * 4f, 0f));
                }
            }

            // stacked 不子 by the door, pointing to 11 船载
            for (int i = 0; i < 8; i++)
            {
                float x = 0.62f + (i % 2) * 0.24f;
                float z = 5.15f + (i / 2) * 0.15f;
                float y = 0.04f + (i / 4) * 0.075f;

            }
            Basket(g, new Vector3(1.35f, 0f, 6.55f), 0.3f, 0.34f, 13);
        }

        static GameObject Basket(Transform parent, Vector3 pos, float r, float h, int seed)
        {
            GameObject go = Group("竹筐", parent, pos, new Vector3(0f, seed * 13f, 0f));
            Vector2[] shell =
            {
                new Vector2(0f, 0f), new Vector2(r * 0.8f, 0f), new Vector2(r, h),
                new Vector2(r * 0.94f, h), new Vector2(r * 0.77f, 0.05f), new Vector2(0f, 0.05f)
            };
            LatheObj("筐体", go.transform, shell, mats.bamboo, Vector3.zero, 18, 1f);
            for (int i = 0; i < 3; i++)
            {
                float y = 0.06f + i * (h - 0.1f) / 2f;
                LatheObj("箍_" + i, go.transform, new[]
                {
                    new Vector2(r * 0.94f, y), new Vector2(r * 1.005f, y),
                    new Vector2(r * 1.005f, y + 0.025f), new Vector2(r * 0.94f, y + 0.025f),
                    new Vector2(r * 0.94f, y)
                }, mats.bamboo, Vector3.zero, 18, 1f);
            }
            for (int i = 0; i < 12; i++)
            {
                float a = i * 30f;
                float rad = a * Mathf.Deg2Rad;
                Vector3 p = new Vector3(Mathf.Cos(rad) * r * 0.9f, h * 0.5f, Mathf.Sin(rad) * r * 0.9f);
                BoxObj("篾_" + i, go.transform, new Vector3(0.014f, h * 0.9f, 0.02f), p,
                    mats.bamboo, new Vector3(0f, -a, 0f), 0.002f, false);
            }
            return go;
        }

            Mesh mesh = ProcMesh.Lathe(shell, 32, 0.8f);
            MeshObj("池体", go.transform, mesh, mats.stone, Vector3.zero, Vector3.zero, Vector3.one);
            BoxCollider mc = go.AddComponent<BoxCollider>();
